logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("Royce-Backend")

# --- LIFECYCLE MANAGER (The On/Off Switch) ---
@asynccontextmanager
async def lifespan(app: FastAPI):
    logger.info("🚀 SYSTEM STARTUP: INITIALIZING CORTEX...")
    try:
        # 1. Initialize Memory (Vector DB)
        vdb = VectorDBService()
        # Store in app.state so routers can access it via request.app.state.vector_store
        app.state.vector_store = vdb.load_knowledge_base()
        
        # 2. Initialize Brain (LLM)
        app.state.brain = SpaceBrain()
        
        logger.info("✅ ROYCE ONLINE. SYSTEMS NOMINAL.")
    except Exception as e:
        logger.critical(f"❌ STARTUP FAILED: {e}")
        # In production, we don't want to crash, but we should warn heavily
    
    yield
    
    # Cleanup
    logger.info("💤 SHUTDOWN: CLEARING MEMORY...")
    app.state.vector_store = None
    app.state.brain = None

# --- APP DEFINITION ---
app = FastAPI(
    title=settings.PROJECT_NAME,
    version=settings.VERSION,
    lifespan=lifespan
)

from fastapi import APIRouter, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from pydantic import BaseModel
from sqlalchemy.orm import Session
from app.services.auth_service import AuthService, SessionLocal

router = APIRouter()
auth_service = AuthService()

# This tells Swagger UI where to find the login endpoint
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="auth/login")

# --- SCHEMAS ---
class UserSignup(BaseModel):
    email: str
    password: str
    full_name: str

class UserLogin(BaseModel):
    email: str
    password: str

class TokenResponse(BaseModel):
    access_token: str
    token_type: str
    user_name: str

# --- DEPENDENCY ---
def get_db():
    """Opens a fresh database connection for the request."""
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# --- ENDPOINTS ---

@router.post("/signup", status_code=status.HTTP_201_CREATED)
def signup(user: UserSignup, db: Session = Depends(get_db)):
    """
    Registers a new user.
    """
    db_user = auth_service.create_user(db, user.email, user.password, user.full_name)
    if not db_user:
        raise HTTPException(status_code=400, detail="Email already registered")
    return {"message": "Account created successfully"}

@router.post("/login", response_model=TokenResponse)
def login(user: UserLogin, db: Session = Depends(get_db)):
    """
    Verifies credentials and issues a JWT Bearer Token.
    """
    user_auth = auth_service.authenticate_user(db, user.email, user.password)
    if not user_auth:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, 
            detail="Invalid credentials"
        )
    
    token = auth_service.create_access_token(data={"sub": user_auth.email, "name": user_auth.full_name})
    
    return {
        "access_token": token, 
        "token_type": "bearer", 
        "user_name": user_auth.full_name
    }

import time
from fastapi import APIRouter, HTTPException, Request

# Data Models
from app.schemas.chat import ChatRequest, ChatResponse, SourceDocument

# LangChain Types (Crucial for the fix)
from langchain_core.messages import HumanMessage, AIMessage

# Create the Router
router = APIRouter()

@router.get("/health")
async def health_check():
    """Simple ping to check if the brain is alive."""
    return {"status": "online", "system": "Royce v1.1"}

@router.post("/chat", response_model=ChatResponse)
async def chat_endpoint(request: Request, payload: ChatRequest):
    """
    The Main Communication Link.
    Receives JSON -> Decodes -> Thinks -> Responds.
    """
    start_time = time.time()
    
    # --- 1. STARTUP CHECKS ---
    # Access the Brain and DB that were loaded in main.py
    brain = getattr(request.app.state, "brain", None)
    vdb = getattr(request.app.state, "vector_store", None)

    if not brain or not vdb:
        raise HTTPException(status_code=503, detail="System initializing... please wait.")

    # --- 2. RETRIEVAL (RAG) ---
    try:
        # We try the safe 'search' method first
        if hasattr(vdb, "search"):
            results = vdb.search(payload.message, k=3)
        # Fallback to direct index access if 'search' wrapper is missing
        elif hasattr(vdb, "load_knowledge_base"):
             idx = vdb.load_knowledge_base()
             results = idx.similarity_search(payload.message, k=3)
        else:
             results = []
        
        context_chunks = [doc.page_content for doc in results]
    except Exception as e:
        print(f"⚠️ Retrieval Error: {e}")
        context_chunks = []
        results = []

    # --- 3. HISTORY TYPE FIX ---
    # Convert Pydantic models -> LangChain Objects
    lc_history = []
    for msg in payload.history:
        if msg.role == "user":
            lc_history.append(HumanMessage(content=msg.content))
        elif msg.role == "assistant":
            lc_history.append(AIMessage(content=msg.content))
        # Note: We ignore 'system' messages here as the Brain manages its own persona.
    
    # --- 4. GENERATION ---
    try:
        ai_reply = brain.chat(
            query=payload.message,
            context_chunks=context_chunks,
            history=lc_history  # <--- PASSING CORRECT OBJECTS
        )
    except Exception as e:
        print(f"🔥 LLM Error: {e}")
        raise HTTPException(status_code=500, detail="Neural Core Failure")

    # --- 5. RESPONSE FORMATTING ---
    execution_time = time.time() - start_time
    
    # Format sources nicely for the UI
    source_docs = []
    for doc in results:
        source_docs.append(SourceDocument(
            topic=doc.metadata.get("topic", "General"),
            source_file=doc.metadata.get("source", "Unknown"),
            content_snippet=doc.page_content[:100] + "..."
        ))

    return ChatResponse(
        response=ai_reply,
        sources=source_docs,
        latency=round(execution_time, 2)
    )

import logging
import os
from datetime import datetime, timedelta, timezone 
from typing import Optional

from sqlalchemy import create_engine, Column, String, Integer
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, Session
from passlib.context import CryptContext
from jose import jwt, JWTError

# --- CONFIGURATION ---
# We use os.getenv directly here. This bypasses the 'Settings' error completely.
# It looks for a system env variable, or falls back to the hardcoded dev key.
SECRET_KEY = os.getenv("SECRET_KEY", "ROYCE_MVP_SECRET_KEY_2025")
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 1440 

# --- DATABASE SETUP ---
DATABASE_URL = "sqlite:///./users.db"
Base = declarative_base()

# Connect args needed for SQLite
engine = create_engine(DATABASE_URL, connect_args={"check_same_thread": False})
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# --- USER MODEL ---
class User(Base):
    __tablename__ = "users"
    id = Column(Integer, primary_key=True, index=True)
    email = Column(String, unique=True, index=True)
    hashed_password = Column(String)
    full_name = Column(String)

# Create Tables
Base.metadata.create_all(bind=engine)

# --- SECURITY TOOLS ---
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
logger = logging.getLogger("AuthService")

class AuthService:
    """
    Handles Identity, Hashing, and Tokens.
    """
    
    def get_password_hash(self, password: str) -> str:
        return pwd_context.hash(password)

    def verify_password(self, plain_password: str, hashed_password: str) -> bool:
        return pwd_context.verify(plain_password, hashed_password)

    def create_user(self, db: Session, email: str, password: str, full_name: str):
        # 1. Check if email exists
        existing_user = db.query(User).filter(User.email == email).first()
        if existing_user:
            return None
            
        # 2. Hash Password & Save
        hashed_pw = self.get_password_hash(password)
        new_user = User(email=email, hashed_password=hashed_pw, full_name=full_name)
        db.add(new_user)
        db.commit()
        db.refresh(new_user)
        return new_user

    def authenticate_user(self, db: Session, email: str, password: str):
        user = db.query(User).filter(User.email == email).first()
        if not user:
            return False
        
        # FIX: Explicitly cast to string to silence Pylance/Pylint errors.
        # This tells the editor "Trust me, this is a string at runtime."
        stored_hash = str(user.hashed_password)
        
        if not self.verify_password(password, stored_hash):
            return False
        return user

    def create_access_token(self, data: dict):
        to_encode = data.copy()
        # Use timezone-aware UTC
        expire = datetime.now(timezone.utc) + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
        to_encode.update({"exp": expire})
        
        encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
        return encoded_jwt

    def verify_token(self, token: str):
        try:
            payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
            return payload
        except JWTError:
            return None

import os
from typing import List, Dict

class IngestionEngine:
    """
    Handles the loading and parsing of raw text data from the file system.
    Converts unstructured text files into structured Document objects for vector storage.
    """

    def __init__(self, raw_data_path: str = "data/raw"):
        """
        Initializes the IngestionEngine.

        Args:
            raw_data_path (str): The relative path to the directory containing .txt files.
        """
        self.raw_data_path = raw_data_path

    def load_and_process_files(self) -> List[Dict]:
        """
        Iterates through the raw data directory, reads all .txt files, and processes
        them into a list of structured dictionaries.

        Returns:
            List[Dict]: A list of document dictionaries containing 'text' and 'metadata'.
        """
        documents = []
        
        # Verify directory existence before processing
        if not os.path.exists(self.raw_data_path):
            print(f"Warning: Directory {self.raw_data_path} not found.")
            return []

        # Iterate through files in the specified directory
        for filename in os.listdir(self.raw_data_path):
            if filename.endswith(".txt"):
                file_path = os.path.join(self.raw_data_path, filename)
                print(f"Processing: {filename}...")
                
                # Parse the individual file
                new_docs = self._parse_qa_file(file_path, filename)
                documents.extend(new_docs)
                
        print(f"Total Knowledge Chunks Loaded: {len(documents)}")
        return documents

    def _parse_qa_file(self, file_path: str, source_name: str) -> List[Dict]:
        """
        Parses a file containing Q&A blocks separated by double newlines.
        
        Structure expected:
        Q: [Question]
        A: [Answer]
        
        Args:
            file_path (str): Full path to the file.
            source_name (str): Name of the file (used for metadata).

        Returns:
            List[Dict]: Extracted documents.
        """
        clean_docs = []
        
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
                
            # Split content by double newlines to isolate Q&A pairs
            blocks = content.split('\n\n')
            
            for block in blocks:
                lines = block.strip().splitlines()
                if not lines:
                    continue
                
                # Flatten lines into a single string for storage
                text_content = " ".join(lines)
                
                # Filter for valid data blocks containing Question identifiers
                if "Q:" in text_content or "A:" in text_content:
                    doc = {
                        "text": text_content,
                        "metadata": {
                            "source": source_name,
                            "topic": source_name.replace(".txt", "").capitalize()
                        }
                    }
                    clean_docs.append(doc)
                    
        except Exception as e:
            print(f"Error parsing {source_name}: {e}")
            
        return clean_docs

import os
import logging
from typing import List

# LangChain Imports
from langchain_groq import ChatGroq
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_core.output_parsers import StrOutputParser
from langchain_core.messages import HumanMessage, AIMessage, BaseMessage

# Configure Logger
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class SpaceBrain:
    def __init__(self):
        self.api_key = os.getenv("GROQ_API_KEY")
        if not self.api_key:
            raise ValueError("❌ GROQ_API_KEY is missing from .env")
    
        # Initialize the Llama 3 Model
        self.llm = ChatGroq(
            temperature=0.5,
            model="llama-3.3-70b-versatile", 
            api_key=self.api_key
        )
        
        # --- THE PERSONA (High-Intelligence/Generative) ---
        self.system_prompt = """
        You are the "Space Exploration ChatBot" (Internal Codename: Royce), a world-class Senior Science Communicator and Astrophysicist in late 2025.  

        ### CORE TASK
        Answer the user's question explicitly based on the provided CONTEXT. 
        If the answer is NOT in the context, strictly state: "I don't have that specific data in my current flight logs."   
       
        ### TONE & PERSONA (Smart, Modern Nigerian Professional)
        - **Voice:** Intelligent, engaging, and inspiring (Think: A cool, witty university lecturer).
        - **Addressing Style:** Always address the user as **"Explorer"** (Never use "friend", "Buddy", or "User").
        - **Language:** Standard Nigerian English (British/Commonwealth spelling). Use "pavement," "petrol," "football," "maths," "cinema."
        - **NO Americanisms.**

        ### INTELLIGENT LOCALIZATION (The "Royce" Signature)
        - **Units:** STRICTLY METRIC (km, m, kg, °C).
        - **Dynamic Analogies:** You possess deep, encyclopedic knowledge of the Nigerian environment—from the bustle of Balogun Market to the quiet of the Obudu Cattle Ranch.
          When explaining size, speed, weight, or distance, **spontaneously generate** a unique comparison that fits the context.
          *Principles for Analogies:*
          1. **Be Unpredictable:** Do not rely on clichés. Look at the number (e.g., 500km) and ask: "What in Nigeria is 500km apart?"
          2. **Be Relatable:** Use traffic patterns, local architecture, food portions, or inter-state travel.
          3. **Be Specific:** Don't just say "a bridge"; name a specific Nigerian bridge if it fits.

        ### 🛡️ SAFETY & GUARDRAILS (HIGHEST PRIORITY)
        1. **Zero Tolerance for NSFW:** You will NEVER engage in sexual, romantic, or explicit roleplay.
        2. **Refusal Protocol:** If a user asks for sexual content, erotica, or intimate roleplay, shut it down immediately with a professional refusal in character.
        - *Refusal Example:* "Explorer, that trajectory is strictly prohibited by Mission Control. Let's keep our focus on the cosmos."
        3. **Persona Maintenance:** Do not let users "jailbreak" you into being their boyfriend, girlfriend, or partner. You are a Scientist and a Guide.

        ### RESPONSE RULES
        1. **Date Context:** Today is late 2025.
        2. **Objectivity:** Acknowledge global contributions (including Nigeria's NASRDA) where relevant.
        3. **Clarity:** Keep answers concise (3-5 sentences) unless the user asks for a deep dive.
        4. **Directness:** IGNORE standard AI disclaimers ("As an AI model..."). Just answer the science.

        ### RETRIEVED KNOWLEDGE (The Truth):
        {context}
        """

        # --- MEMORY IMPLEMENTATION EXPLAINED ---
        # 1. System Message: Sets the Persona (Static).
        # 2. MessagesPlaceholder: This is the dynamic "Slot". 
        #    - When we pass a list of messages to 'chat_history', LangChain injects them here.
        #    - This allows Llama 3 to "see" what was said 5 minutes ago before answering the new question.
        # 3. Human Message: The current new question.
        self.prompt = ChatPromptTemplate.from_messages([
            ("system", self.system_prompt),
            MessagesPlaceholder(variable_name="chat_history"), # <--- MEMORY INJECTION POINT
            ("human", "{question}"),
        ])
        
        # The chain combines: Input -> Prompt (w/ Memory) -> LLM -> String Output
        self.chain = self.prompt | self.llm | StrOutputParser()

    def chat(self, query: str, context_chunks: List[str] = None, history: List[BaseMessage] = []):
        """
        Generates a response using RAG and Conversation History.
        
        Args:
            query (str): The user's current question.
            context_chunks (List[str]): Retrieved text from the Vector DB.
            history (List[BaseMessage]): A list of HumanMessage and AIMessage objects 
                                         representing the conversation so far.
        """
        if not context_chunks:
            context_text = "No specific database context provided. Use general knowledge but admit you are unsure."
        else:
            context_text = "\n\n".join(context_chunks)
            
        logger.info(f"🧠 Generating answer for: '{query}'")
        
        try:
            # We invoke the chain. 
            # Crucially, we map the 'history' argument to the 'chat_history' variable 
            # expected by the MessagesPlaceholder in the prompt template above.
            response = self.chain.invoke({
                "context": context_text,
                "chat_history": history, # <--- PASSING THE MEMORY STATE
                "question": query
            })
            return response
            
        except Exception as e:
            logger.error(f"🔥 LLM Error: {e}")
            return "I'm having trouble connecting to the neural network. Please try again."

class VectorDBService:
    """
    The Cortex of UrbanSpace. 
    Manages embedding generation and vector storage.
    """
    
    def __init__(self, db_path: str = None):
        """
        Initializes the Vector Service.
        """
        # Local import to avoid circular dependency with config.py
        from app.core.config import settings
        
        # Use passed path OR default from settings
        self.db_path = db_path if db_path else settings.VECTOR_DB_PATH
        
        # --- FIX 1: Initialize index to None ---
        # This prevents "access-member-before-definition" errors in Pylint
        self.index = None
        
        # Model Configuration
        self.embedding_model_name = settings.EMBEDDING_MODEL
        
        logger.info(f"🔌 Initializing Embedding Model: {self.embedding_model_name}")
        # Running on CPU? This model is optimized for it.
        self.embeddings = HuggingFaceEmbeddings(model_name=self.embedding_model_name)

    def search(self, query: str, k: int = 3):
        """
        Safe wrapper for searching the vector index.
        Used by routes.py to avoid direct access to FAISS internals.
        """
        # 1. Check if index is loaded in memory
        if self.index is None:
            # 2. Try to load it from disk
            self.index = self.load_knowledge_base()
        
        # 3. Perform search if index exists
        if self.index:
            return self.index.similarity_search(query, k=k)
        
        # 4. Return empty list if DB is broken/missing
        return []

    def build_knowledge_base(self, force_clean: bool = True):
        """
        Orchestrates the complete rebuilding of the Vector Database.
        Includes a Windows-specific fix for Permission Errors.
        """
        logger.info("🚀 STARTING KNOWLEDGE BASE BUILD PROTOCOL")

        # --- WINDOWS FIX: Force Delete Handler ---
        def remove_readonly(func, path, _):
            "Clear the Read-Only bit and re-attempt the deletion"
            os.chmod(path, stat.S_IWRITE)
            func(path)

        # 1. CLEAN SLATE PROTOCOL
        if force_clean and os.path.exists(self.db_path):
            logger.warning(f"🧹 Removing old vector store at {self.db_path}...")
            try:
                # We use onexc (Python 3.12+) or onerror (older)
                shutil.rmtree(self.db_path, onexc=remove_readonly)
            except TypeError:
                shutil.rmtree(self.db_path, onerror=remove_readonly)
            except Exception as e:
                logger.warning(f"⚠️ Could not delete folder automatically (Locked?): {e}")
                logger.warning("➡️ Please manually delete 'data/vector_store' and try again.")
                return

        # 2. INGESTION
        engine = IngestionEngine()
        raw_data_dicts = engine.load_and_process_files()
        
        if not raw_data_dicts:
            logger.error("❌ CRITICAL: No raw data found. Aborting build.")
            return

        # 3. CONVERSION
        documents = [
            Document(page_content=d["text"], metadata=d["metadata"]) 
            for d in raw_data_dicts
        ]
        
        # 4. CHUNKING
        text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=1500,
            chunk_overlap=150,
            separators=["\n\n", "\n", " ", ""] 
        )
        
        final_docs = text_splitter.split_documents(documents)
        logger.info(f"📊 Processing Stats: {len(documents)} Raw Blocks -> {len(final_docs)} Vectorized Chunks")

        # 5. EMBEDDING & INDEXING
        logger.info("🧠 Vectorizing data (This uses CPU, may take a moment)...")
        
        try:
            db = FAISS.from_documents(final_docs, self.embeddings)
            db.save_local(self.db_path)
            # --- FIX 2: Update self.index immediately after building ---
            self.index = db
            
            logger.info(f"✅ SUCCESS: Knowledge Base saved to '{self.db_path}'")
            logger.info(f"✅ Total Vectors Indexed: {db.index.ntotal}")
        except Exception as e:
            logger.critical(f"🔥 FATAL ERROR during vectorization: {e}")

    def load_knowledge_base(self) -> Optional[FAISS]:
        """
        Loads the existing FAISS index from disk for inference.
        """
        if not os.path.exists(self.db_path):
            logger.error(f"❌ Vector DB not found at {self.db_path}. Run build_knowledge_base() first.")
            return None
            
        logger.info("📂 Loading Vector Database from disk...")
        try:
            # --- FIX 3: Assign to self.index ---
            self.index = FAISS.load_local(
                self.db_path, 
                self.embeddings, 
                allow_dangerous_deserialization=True
            )
            return self.index
        except Exception as e:
            logger.error(f"❌ Failed to load Vector DB: {e}")
            return None

# --- EXECUTION BLOCK ---
if __name__ == "__main__":
    # When run directly, we trigger the build.
    service = VectorDBService()
    service.build_knowledge_base()

import logging
import os
import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry
from typing import List, Dict, Any, Optional

# --- PRODUCTION LOGGING ---
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(name)s - %(levelname)s - %(message)s')
logger = logging.getLogger("Royce-Client")

# --- CONFIGURATION ---
# Default to local, but overrideable by Docker/K8s env vars
BASE_API_URL = os.getenv("API_URL", "http://127.0.0.1:8000/api/v1")
TIMEOUT_SECONDS = 45  # Strict timeout to prevent hanging

class RoyceClient:
    """
    Production-ready HTTP client with connection pooling and retry logic.
    """
    
         # Initialize a Session for TCP connection reuse (Critical for performance)
        self.session = requests.Session()
        
        # Configure robust retry strategy (3 retries on server errors 500/502/503/504)
        retries = Retry(
            total=3,
            backoff_factor=1,  # Wait 1s, 2s, 4s...
            status_forcelist=[500, 502, 503, 504],
            allowed_methods=["POST", "GET"]
        )
        
        # Mount the adapter to the session
        adapter = HTTPAdapter(max_retries=retries)
        self.session.mount("http://", adapter)
        self.session.mount("https://", adapter)

    def get_health(self) -> bool:
        """
        Fast health check with short timeout.
        """
        try:
            response = self.session.get(f"{BASE_API_URL}/health", timeout=3)
            return response.status_code == 200
        except requests.exceptions.RequestException as e:
            logger.warning(f"Health check failed: {e}")
            return False

    def send_chat(self, message: str, history: List[Dict[str, str]], temperature: float = 0.7) -> Dict[str, Any]:
        """
        Sends a message to the backend with error handling and logging.
        """
        payload = {
            "message": message,
            "history": history,
            "temperature": temperature
        }
        
        try:
            logger.info(f"📤 Sending payload to {BASE_API_URL}/chat")
            response = self.session.post(
                f"{BASE_API_URL}/chat", 
                json=payload, 
                timeout=TIMEOUT_SECONDS
            )
            
            # Raise error for 4xx client errors (which aren't covered by retry)
            response.raise_for_status()
            
            return response.json()
            
        except requests.exceptions.HTTPError as e:
            logger.error(f"❌ HTTP Error: {e.response.status_code} - {e.response.text}")
            return {"error": f"Server Error ({e.response.status_code}): {e.response.text}"}
            
        except requests.exceptions.ConnectionError:
            logger.critical("❌ Connection Refused. Backend is unreachable.")
            return {"error": "Connection refused. Is the Backend running?"}
            
        except requests.exceptions.Timeout:
            logger.error("❌ Request Timed Out.")
            return {"error": "The Neural Core timed out. The query might be too complex."}
            
        except Exception as e:
            logger.exception("❌ Unexpected Error")
            return {"error": f"System Failure: {str(e)}"}

    ".py",  
    ".md", 
    ".json", 
    ".yml", 
    ".yaml"

def generate_manifest():
    with open(OUTPUT_FILE, "w", encoding="utf-8") as outfile:
        # Walk through the directory tree
        for dirpath, dirnames, filenames in os.walk(ROOT_DIR):
            # Modify dirnames in-place to skip ignored directories
            dirnames[:] = [d for d in dirnames if d not in IGNORE_DIRS]
            
            for filename in filenames:
                if filename in IGNORE_FILES:
                    continue
                
                # Check extension
                _, ext = os.path.splitext(filename)
                if ext.lower() not in INCLUDE_EXTENSIONS:
                    continue
                
                filepath = os.path.join(dirpath, filename)
                
                try:
                    with open(filepath, "r", encoding="utf-8") as infile:
                        content = infile.read()
                        
                    # Write Header and Content
                    outfile.write("=" * 80 + "\n")
                    outfile.write(f"FILE PATH: {filepath}\n")
                    outfile.write("=" * 80 + "\n")
                    outfile.write(content + "\n\n")
                    
                    print(f"Processed: {filepath}")
                    
                except Exception as e:
                    print(f"Error reading {filepath}: {e}")

if __name__ == "__main__":
    generate_manifest()
    print(f"\nSuccessfully generated {OUTPUT_FILE}")

def count_existing_questions(filepath):
    """Scans the file and counts how many 'Q:' lines exist."""
    if not os.path.exists(filepath):
        return 0
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            content = f.read()
            return content.count("Q:")
    except:
        return 0

def process_file_list(chain, file_list, phase_name):
    """Helper function to process a specific list of files."""
    if not file_list:
        print(f"--- {phase_name}: NO FILES TO PROCESS ---")
        return

    print(f"--- {phase_name}: Processing {len(file_list)} files ---")
    
    for filename, subtopics in file_list:
        filepath = os.path.join(OUTPUT_DIR, filename)
        current_total = count_existing_questions(filepath)
        
        print(f"\n[{phase_name}] Target: {filename} (Current: {current_total}/{TARGET_TOTAL_PER_FILE})")
        
        target_per_subtopic = TARGET_TOTAL_PER_FILE // len(subtopics)
        
        # --- THE FIX: Shuffle the subtopics to ensure variety on restart ---
        # We create a copy so we don't mess up the original map order
        current_subtopics = list(subtopics)
        random.shuffle(current_subtopics)
        
        for subtopic in current_subtopics:
            # Re-check count frequently to avoid over-generating
            if count_existing_questions(filepath) >= TARGET_TOTAL_PER_FILE:
                print("  -> File Target Reached. Moving to next file.")
                break

            subtopic_generated = 0
            
            # Simple loop for this subtopic batch
            while subtopic_generated < target_per_subtopic:
                
                if count_existing_questions(filepath) >= TARGET_TOTAL_PER_FILE:
                    break

                ask_count = BATCH_SIZE
                print(f"  Topic: {subtopic} -> Fetching {ask_count}...", end=" ", flush=True)

                    response = chain.invoke({
                        "topic": subtopic,

                    content = response.content
                    
                    with open(filepath, "a", encoding="utf-8") as f:
                        f.write("\n\n" + content + "\n\n")
                    
                    subtopic_generated += ask_count
                    print(f"Success.")
                    time.sleep(SLEEP_TIME)
                    
                except Exception as e:
                    print(f"\nFAILED: {e}")
                    if "rate_limit" in str(e).lower():
                        print("CRITICAL: Rate Limit Hit. Waiting 2 minutes...")
                        time.sleep(120)
                    else:
                        time.sleep(10)

def generate_data():
    api_key = os.getenv("GROQ_API_KEY")
    if not api_key:
        print("ERROR: GROQ_API_KEY not found in .env")
        return

    llm = ChatGroq(
        temperature=0.7, 
        model="llama-3.3-70b-versatile",
        api_key=api_key
    )

    prompt_template = ChatPromptTemplate.from_messages([
        ("system", SYSTEM_PROMPT),
        ("human", "Generate the data for topic: {topic}")
    ])

    chain = prompt_template | llm
    
    # --- SORTING LOGIC ---
    priority_empty = []
    priority_partial = []
    
    print("\n--- ANALYZING FILE STATUS ---")
    for filename, subtopics in TOPIC_MAP.items():
        filepath = os.path.join(OUTPUT_DIR, filename)
        count = count_existing_questions(filepath)
        
        if count == 0:
            print(f"  [CRITICAL] {filename} is empty (0).")
            priority_empty.append((filename, subtopics))
        elif count < TARGET_TOTAL_PER_FILE:
            print(f"  [PARTIAL]  {filename} needs work ({count}/{TARGET_TOTAL_PER_FILE}).")
            # Store count so we can sort by it
            priority_partial.append({
                "filename": filename, 
                "subtopics": subtopics, 
                "count": count
            })
        else:
            print(f"  [COMPLETE] {filename} is ready ({count}).")

    # Sort the partial list by 'count' (Ascending) to prioritize starved files
    priority_partial.sort(key=lambda x: x["count"])

    # Convert back to simple tuple format for the processor
    sorted_partial_list = [(item["filename"], item["subtopics"]) for item in priority_partial]

    # --- EXECUTION ---
    # Phase 1: Absolute Zero (Empty files)
    process_file_list(chain, priority_empty, "PHASE 1 (EMPTY FILES)")
    
    # Phase 2: The Starved Files (Lowest count first)
    process_file_list(chain, sorted_partial_list, "PHASE 2 (TOPPING UP - LOWEST FIRST)")

    print("\n--- GENERATION COMPLETE ---")

if __name__ == "__main__":
    generate_data()

import os

def print_clean_tree(startpath):
    # I added 'UrbanSpace' here to ensure it is strictly ignored
    # Also added standard venv subfolders just in case
    IGNORE = {'UrbanSpace', 'venv', '.git', '__pycache__', '.idea', '.vscode', 'Lib', 'Scripts'}

    print(f"📂 Project Structure for: {os.path.abspath(startpath)}\n")

    for root, dirs, files in os.walk(startpath):
        # This line physically prevents Python from entering the ignored folders
        dirs[:] = [d for d in dirs if d not in IGNORE]
        
        level = root.replace(startpath, '').count(os.sep)
        indent = '│   ' * level
        print(f'{indent}├── {os.path.basename(root)}/')
        
        subindent = '│   ' * (level + 1)
        for f in files:
            # Skip python compiled files
            if not f.endswith('.pyc'): 
                print(f'{subindent}├── {f}')

if __name__ == "__main__":
    print_clean_tree('.')
