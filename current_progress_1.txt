import os

# Define the target
TARGET = 500

def format_time(timestamp):
    """Formats a timestamp for professional logging."""
    return datetime.fromtimestamp(timestamp).strftime("%H:%M:%S")

def count_existing_questions(filepath):
    """Scans the file and counts how many 'Q:' lines exist."""
    if not os.path.exists(filepath):
        return 0
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            content = f.read()
            return content.count("Q:")
    except:
        return 0

def verify_data():

    start_time = time.time()

            # Print row for the main table

        except Exception as e:
            print(f"{filename:<25} | âŒ ERROR    | Failed to read file: {str(e)}")
            all_pass = False
            
    end_time = time.time()

import os
from typing import List, Dict

class IngestionEngine:
    """
    Handles the loading and parsing of raw text data from the file system.
    Converts unstructured text files into structured Document objects for vector storage.
    """

    def __init__(self, raw_data_path: str = "data/raw"):
        """
        Initializes the IngestionEngine.

        Args:
            raw_data_path (str): The relative path to the directory containing .txt files.
        """
        self.raw_data_path = raw_data_path

    def load_and_process_files(self) -> List[Dict]:
        """
        Iterates through the raw data directory, reads all .txt files, and processes
        them into a list of structured dictionaries.

        Returns:
            List[Dict]: A list of document dictionaries containing 'text' and 'metadata'.
        """
        documents = []
        
        # Verify directory existence before processing
        if not os.path.exists(self.raw_data_path):
            print(f"Warning: Directory {self.raw_data_path} not found.")
            return []

        # Iterate through files in the specified directory
        for filename in os.listdir(self.raw_data_path):
            if filename.endswith(".txt"):
                file_path = os.path.join(self.raw_data_path, filename)
                print(f"Processing: {filename}...")
                
                # Parse the individual file
                new_docs = self._parse_qa_file(file_path, filename)
                documents.extend(new_docs)
                
        print(f"Total Knowledge Chunks Loaded: {len(documents)}")
        return documents

    def _parse_qa_file(self, file_path: str, source_name: str) -> List[Dict]:
        """
        Parses a file containing Q&A blocks separated by double newlines.
        
        Structure expected:
        Q: [Question]
        A: [Answer]
        
        Args:
            file_path (str): Full path to the file.
            source_name (str): Name of the file (used for metadata).

        Returns:
            List[Dict]: Extracted documents.
        """
        clean_docs = []
        
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
                
            # Split content by double newlines to isolate Q&A pairs
            blocks = content.split('\n\n')
            
            for block in blocks:
                lines = block.strip().splitlines()
                if not lines:
                    continue
                
                # Flatten lines into a single string for storage
                text_content = " ".join(lines)
                
                # Filter for valid data blocks containing Question identifiers
                if "Q:" in text_content or "A:" in text_content:
                    doc = {
                        "text": text_content,
                        "metadata": {
                            "source": source_name,
                            "topic": source_name.replace(".txt", "").capitalize()
                        }
                    }
                    clean_docs.append(doc)
                    
        except Exception as e:
            print(f"Error parsing {source_name}: {e}")
            
        return clean_docs

# --- Execution Entry Point (For testing module in isolation) ---

import os
from langchain_community.vectorstores import FAISS
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from app.services.ingestion import IngestionEngine

class VectorDBService:
    """
    Manages the long-term memory of the AI.
    Converts text -> Vectors -> FAISS Database.
    """
    
    def __init__(self, db_path="data/vector_store"):
        self.db_path = db_path
        # We use 'all-MiniLM-L6-v2'. It is fast, free, and runs on your laptop CPU.
        # It creates a 384-dimensional vector for every sentence.
        print("--- Loading Embedding Model (all-MiniLM-L6-v2) ---")
        self.embeddings = HuggingFaceEmbeddings(model_name="all-MiniLM-L6-v2")

    def build_knowledge_base(self):
        """
        Runs the Ingestion Engine to read .txt files,
        chunks them, and builds a fresh FAISS index.
        """
        print("\n--- STARTING MEMORY BUILD ---")
        
        # 1. Ingest Data
        engine = IngestionEngine()
        raw_docs = engine.load_and_process_files()
        
        if not raw_docs:
            print("CRITICAL: No data found to ingest! Check data/raw folder.")
            return

        # 2. Chunking
        # We use 700 characters per chunk with 70 overlap.
        # This ensures the AI gets enough context (about 3-5 sentences) per search result.
        text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=700,
            chunk_overlap=70
        )
        
        docs = text_splitter.create_documents(
            [d['text'] for d in raw_docs], 
            metadatas=[d['metadata'] for d in raw_docs]
        )
        print(f"  -> Split {len(raw_docs)} raw blocks into {len(docs)} searchable vectors.")

        # 3. Create Vector Store
        print("  -> Vectorizing data... (This uses your CPU, please wait...)")
        db = FAISS.from_documents(docs, self.embeddings)

        # 4. Save to Disk
        # This creates the 'data/vector_store' folder
        db.save_local(self.db_path)
        print(f"SUCCESS: Knowledge Base built and saved to {self.db_path}")

    def load_knowledge_base(self):
        """
        Loads the existing FAISS index from disk.
        """
        if not os.path.exists(self.db_path):
            raise ValueError(f"Vector DB not found at {self.db_path}. Run build_knowledge_base() first.")
            
        return FAISS.load_local(
            self.db_path, 
            self.embeddings, 
            allow_dangerous_deserialization=True
        )

if __name__ == "__main__":
    # When run directly, it rebuilds the database
    vdb = VectorDBService()
    vdb.build_knowledge_base()

def count_existing_questions(filepath):
    """Scans the file and counts how many 'Q:' lines exist."""
    if not os.path.exists(filepath):
        return 0
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            content = f.read()
            return content.count("Q:")
    except:
        return 0

def process_file_list(chain, file_list, phase_name):
    """Helper function to process a specific list of files."""
    if not file_list:
        print(f"--- {phase_name}: NO FILES TO PROCESS ---")
        return

    print(f"--- {phase_name}: Processing {len(file_list)} files ---")
    
    for filename, subtopics in file_list:
        filepath = os.path.join(OUTPUT_DIR, filename)
        current_total = count_existing_questions(filepath)
        
        print(f"\n[{phase_name}] Target: {filename} (Current: {current_total}/{TARGET_TOTAL_PER_FILE})")
        
        target_per_subtopic = TARGET_TOTAL_PER_FILE // len(subtopics)
        
        # --- THE FIX: Shuffle the subtopics to ensure variety on restart ---
        # We create a copy so we don't mess up the original map order
        current_subtopics = list(subtopics)
        random.shuffle(current_subtopics)
        
        for subtopic in current_subtopics:
            # Re-check count frequently to avoid over-generating
            if count_existing_questions(filepath) >= TARGET_TOTAL_PER_FILE:
                print("  -> File Target Reached. Moving to next file.")
                break

            subtopic_generated = 0
            
            # Simple loop for this subtopic batch
            while subtopic_generated < target_per_subtopic:
                
                if count_existing_questions(filepath) >= TARGET_TOTAL_PER_FILE:
                    break

                ask_count = BATCH_SIZE
                print(f"  Topic: {subtopic} -> Fetching {ask_count}...", end=" ", flush=True)
                
                try:
                    response = chain.invoke({
                        "topic": subtopic,
                        "count": str(ask_count)
                    })
                    
                    content = response.content
                    
                    with open(filepath, "a", encoding="utf-8") as f:
                        f.write("\n\n" + content + "\n\n")
                    
                    subtopic_generated += ask_count
                    print(f"Success.")
                    time.sleep(SLEEP_TIME)
                    
                except Exception as e:
                    print(f"\nFAILED: {e}")
                    if "rate_limit" in str(e).lower():
                        print("CRITICAL: Rate Limit Hit. Waiting 2 minutes...")
                        time.sleep(120)
                    else:
                        time.sleep(10)

def generate_data():
    api_key = os.getenv("GROQ_API_KEY")
    if not api_key:
        print("ERROR: GROQ_API_KEY not found in .env")
        return

    llm = ChatGroq(
        temperature=0.7, 
        model="llama-3.3-70b-versatile",
        api_key=api_key
    )

    prompt_template = ChatPromptTemplate.from_messages([
        ("system", SYSTEM_PROMPT),
        ("human", "Generate the data for topic: {topic}")
    ])

    chain = prompt_template | llm
    
    # --- SORTING LOGIC ---
    priority_empty = []
    priority_partial = []
    
    print("\n--- ANALYZING FILE STATUS ---")
    for filename, subtopics in TOPIC_MAP.items():
        filepath = os.path.join(OUTPUT_DIR, filename)
        count = count_existing_questions(filepath)
        
        if count == 0:
            print(f"  [CRITICAL] {filename} is empty (0).")
            priority_empty.append((filename, subtopics))
        elif count < TARGET_TOTAL_PER_FILE:
            print(f"  [PARTIAL]  {filename} needs work ({count}/{TARGET_TOTAL_PER_FILE}).")
            # Store count so we can sort by it
            priority_partial.append({
                "filename": filename, 
                "subtopics": subtopics, 
                "count": count
            })
        else:
            print(f"  [COMPLETE] {filename} is ready ({count}).")

    # Sort the partial list by 'count' (Ascending) to prioritize starved files
    priority_partial.sort(key=lambda x: x["count"])

    # Convert back to simple tuple format for the processor
    sorted_partial_list = [(item["filename"], item["subtopics"]) for item in priority_partial]

    # --- EXECUTION ---
    # Phase 1: Absolute Zero (Empty files)
    process_file_list(chain, priority_empty, "PHASE 1 (EMPTY FILES)")
    
    # Phase 2: The Starved Files (Lowest count first)
    process_file_list(chain, sorted_partial_list, "PHASE 2 (TOPPING UP - LOWEST FIRST)")

    print("\n--- GENERATION COMPLETE ---")

if __name__ == "__main__":
    generate_data()

import sys

from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

from knowledge_base  import load_knowledge_base
from utils import preprocess_text, playful_responses

class Chatbot:
    def __init__(self, knowledge_base_path: str, confidence_threshold: float = 0.2):
        """
        Initialise the chatbot:
        - Load the knowledge base 
        - Preprocess all questions
        - Build TF-IDF matrix for  similarity search
        - Set a configurable confidence threshold
        """

        self.knowledge_base = load_knowledge_base(knowledge_base_path)
        self.confidence_threshold = confidence_threshold

        # Separate questions and answers 
        self.questions = [item["question"] for item in self.knowledge_base]
        self.answers =  [item["answer"] for item in self.knowledge_base]


        # Build TF_IDF vectorizer
        self.vectorizer = TfidfVectorizer(tokenizer= preprocess_text)
        self.question_vectors = self.vectorizer.fit_transform(self.processed_questions)

    def get_response(self, user_input: str) -> str:
        """
        Process user input, compare it with knowledge base
        and returns the most relevant answer.
        """

        if not user_input.strip():
            return playful_responses()
       
        processed_input = preprocess_text(user_input)
        input_vector = self.vectorizer.transform([processed_input])

        # Compute Similarity
        similarities = cosine_similarity(input_vector, self.question_vectors)
        best_match_index = similarities.argmax()
        best_score = similarities[0][best_match_index]

        # Apply configurable confidence threshold
        if best_score < self.confidence_threshold:
            return playful_responses()
        return self.answers[best_match_index]

import os

def load_knowledge_base(directory_path):
    """
    Loads all .txt files from the given directory, parses Q/A pairs, and returns a list of dicts:
    [
        {"question": "...", "answer": "..."},
        ...
    ]
    Assumes each Q/A pair in the file is formatted as:
    Q: ...\nA: ...\n (blank line between pairs)
    """

    knowledge = []
    for filename in os.listdir(directory_path):
        if filename.endswith(".txt"):
            file_path = os.path.join(directory_path, filename)  # Full path to file
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
            # Split into blocks by blank lines
            blocks = [block.strip() for block in content.split('\n\n') if block.strip()]
            for block in blocks:
                lines = block.splitlines()
                question, answer = None, None
                for line in lines:
                    if line.startswith('Q:'):
                        question = line[2:].strip()
                    elif line.startswith('A:'):
                        answer = line[2:].strip()
                if question and answer:
                    knowledge.append({"question": question, "answer": answer})
    return knowledge

def preprocess_text(text):
    """
    Tokenizes, lowercases, removes punctuation, and lemmatizes input text.add
    Returns a list of clean tokens. 
    """
    # initializes lemmatizer 
    lemmatizer = WordNetLemmatizer()

    # Step 1: Tokenize 
    tokens = nltk.word_tokenize(text)

    # Step 2: Lowercase
    tokens = [token.lower() for token in tokens]

    # Step 3: Remove Punctuation
    tokens = [token for token in tokens if token not in string.punctuation]

    # Step 4: POS (Part of speech) tagging 
    pos_tags = nltk.pos_tag(tokens)

    # Steps 5: Lemmatize based on POS
    def get_wordnet_pos(tag):
        """
        Convert POS tag to format recognized by WordNetLemmatizer
        """
        if tag.startswith('J'):
            return wordnet.ADJ
        elif tag.startswith('V'):
            return wordnet.VERB
        elif tag.startswith('N'):
            return wordnet.NOUN
        elif tag.startswith('R'):
            return wordnet.ADV
        else:
            return wordnet.NOUN    # default

    lemmatized_tokens = [lemmatizer.lemmatize(word, get_wordnet_pos(tag)) for word, tag in pos_tags]

    return lemmatized_tokens
